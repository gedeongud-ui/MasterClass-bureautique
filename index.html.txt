        isFounder: true
      }
    };
    localStorage.setItem(STORAGE_CODES_KEY, JSON.stringify(codes));
  }
  if (!localStorage.getItem(STORAGE_USERS_KEY)) {
    localStorage.setItem(STORAGE_USERS_KEY, JSON.stringify({}));
  }
}

function switchAuthTab(tab) {
  document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
  document.getElementById(tab === 'student' ? 'tab-btn-student' : 'tab-btn-creator').classList.add('active');
  document.getElementById('student-form').style.display = tab === 'student' ? 'block' : 'none';

function showAlert(msg, type = 'error') {
  const el = document.getElementById('auth-alert');
  el.textContent = msg;
  el.className = 'alert ' + (type === 'error' ? 'alert-error' : 'alert-success');
  el.style.display = 'block';
}
function hideAlert() { document.getElementById('auth-alert').style.display = 'none'; }

function unlockCreator() {
  const pin = document.getElementById('creator-master-pwd').value.trim();
  if (pin !== CREATOR_PIN) {
    showAlert("Mot de passe maître du Créateur incorrect.");
    return;
  }
  hideAlert();
  document.getElementById('creator-tools').style.display = 'block';
  renderCodesTable();
}

function acceptManualCode() {
  const rawCode = document.getElementById('manual-code-input').value.trim();
  if (!rawCode) {
    showAlert("Veuillez saisir le code que vous souhaitez accepter (ex: 20262).");
    return;
  }
  const codes = JSON.parse(localStorage.getItem(STORAGE_CODES_KEY));
  if (codes[rawCode]) {
    showAlert(`Le code ${rawCode} est déjà enregistré dans la liste.`);
    return;
  }
  const assignee = document.getElementById('new-code-assignee').value.trim() || "Élève autorisé par le Créateur";
  codes[rawCode] = { code: rawCode, assignedTo: assignee, claimedBy: null, claimedName: null, isFounder: false };
  localStorage.setItem(STORAGE_CODES_KEY, JSON.stringify(codes));
  document.getElementById('manual-code-input').value = "";
  document.getElementById('new-code-assignee').value = "";
  renderCodesTable();
  showAlert(`✅ Code ${rawCode} accepté et activé pour : ${assignee}`, 'success');
}

function generateSequentialCode() {
  const codes = JSON.parse(localStorage.getItem(STORAGE_CODES_KEY));
  let maxN = 1;
  Object.keys(codes).forEach(k => {
    const n = parseInt(k.replace("2026", ""), 10);
    if (!isNaN(n) && n > maxN) maxN = n;
  });
  const nextCode = "2026" + (maxN + 1);
  const assignee = document.getElementById('new-code-assignee').value.trim() || `Élève ${maxN}`;
  codes[nextCode] = { code: nextCode, assignedTo: assignee, claimedBy: null, claimedName: null, isFounder: false };
  localStorage.setItem(STORAGE_CODES_KEY, JSON.stringify(codes));
  document.getElementById('new-code-assignee').value = "";
  renderCodesTable();
  showAlert(`✅ Nouveau code unique ${nextCode} généré et accepté pour : ${assignee}`, 'success');
}

function revokeCode(codeStr) {
  if (codeStr === FOUNDER_CODE) {
    alert("Le code fondateur 20261 ne peut pas être révoqué.");
    return;
  }
  const codes = JSON.parse(localStorage.getItem(STORAGE_CODES_KEY));
  delete codes[codeStr];
  localStorage.setItem(STORAGE_CODES_KEY, JSON.stringify(codes));
  renderCodesTable();
}

function renderCodesTable() {
  const codes = JSON.parse(localStorage.getItem(STORAGE_CODES_KEY));
  const tbody = document.getElementById('codes-table-body');
  if (!tbody) return;
  tbody.innerHTML = Object.values(codes).map(c => `
    <tr>
      <td><strong>${c.code}</strong> ${c.isFounder ? '👑' : ''}</td>
      <td>${c.claimedName ? c.claimedName : c.assignedTo}</td>
      <td>${c.claimedBy ? '<span style="color:#b91c1c;font-weight:700;">🔒 Activé (1 seul titulaire)</span>' : '<span style="color:#047857;font-weight:700;">✅ Accepté / En attente</span>'}</td>
      <td>
        <button type="button" class="btn btn-sm btn-outline" onclick="navigator.clipboard?.writeText('${c.code}'); alert('Code ${c.code} copié !')">Copier</button>
        ${!c.isFounder ? `<button type="button" class="btn btn-sm btn-danger" onclick="revokeCode('${c.code}')">Retirer</button>` : ''}
      </td>
    </tr>
  `).join('');
}

let CURRENT_USER = null;

function handleStudentLogin(e) {
  e.preventDefault();
  const nom = document.getElementById('st-nom').value.trim().toUpperCase();
  const prenom = document.getElementById('st-prenom').value.trim();
  const personalPwd = document.getElementById('st-personal-pwd').value.trim();
  const creatorCode = document.getElementById('st-creator-code').value.trim();
  const ageGroup = document.getElementById('st-age').value;

  const codes = JSON.parse(localStorage.getItem(STORAGE_CODES_KEY));
  const users = JSON.parse(localStorage.getItem(STORAGE_USERS_KEY));

  // Vérification stricte : le code doit avoir été accepté par le créateur
  if (!codes[creatorCode]) {
    showAlert(`Accès refusé : Le code "${creatorCode}" n'a pas été accepté par le Créateur. Seuls les codes validés manuellement dans l'Espace Créateur fonctionnent.`);
    return;
  }

  const userKey = `${nom}_${prenom.toLowerCase()}_${creatorCode}`;
  const codeRecord = codes[creatorCode];

  // Si le code est déjà revendiqué par un autre utilisateur
  if (codeRecord.claimedBy && codeRecord.claimedBy !== userKey) {
    showAlert(`Sécurité : Le code ${creatorCode} est déjà exclusivement attribué à ${codeRecord.claimedName}. Une seule personne peut utiliser un code.`);
    return;
  }

  // Reprise de compte existant avec mot de passe personnel
  if (users[userKey]) {
    if (users[userKey].personalPwd !== personalPwd) {
      showAlert("Mot de passe personnel incorrect pour ce compte.");
      return;
    }
    const today = new Date().toISOString().slice(0, 10);
    if (!users[userKey].studyDates.includes(today)) {
      users[userKey].studyDates.push(today);
    }
    users[userKey].ageGroup = ageGroup;
    CURRENT_USER = users[userKey];
  } else {
    // Première activation du code par son titulaire légitime
    const today = new Date().toISOString().slice(0, 10);
    codeRecord.claimedBy = userKey;
    codeRecord.claimedName = `${prenom} ${nom}`;
    localStorage.setItem(STORAGE_CODES_KEY, JSON.stringify(codes));

    CURRENT_USER = {
      userKey,
      nom,
      prenom,
      personalPwd,
      creatorCode,
      isFounder: creatorCode === FOUNDER_CODE,
      ageGroup,
      studyDates: [today],
      completedItems: {},
      examScores: { windows: 0, word: 0, excel: 0, ppt: 0, access: 0 },
      qcmScore: 0,
      qcmDone: false,
      founderSig: "Blessing TSIOBINDA"
    };
    users[userKey] = CURRENT_USER;
  }

  localStorage.setItem(STORAGE_USERS_KEY, JSON.stringify(users));
  localStorage.setItem(SESSION_KEY, userKey);
  launchApp();
}

function saveUserProgress() {
  if (!CURRENT_USER) return;
  const users = JSON.parse(localStorage.getItem(STORAGE_USERS_KEY));
  users[CURRENT_USER.userKey] = CURRENT_USER;
  localStorage.setItem(STORAGE_USERS_KEY, JSON.stringify(users));
  updateTopStats();
}

  localStorage.removeItem(SESSION_KEY);
  CURRENT_USER = null;
  document.getElementById('app-screen').style.display = 'none';
  document.getElementById('coach-widget').style.display = 'none';
  document.getElementById('auth-screen').style.display = 'flex';
}

function exportUserBackup() {
  if (!CURRENT_USER) return;
  const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(CURRENT_USER, null, 2));
  const a = document.createElement('a');

  a.setAttribute("download", `Sauvegarde_${CURRENT_USER.nom}_${CURRENT_USER.creatorCode}.json`);
  document.body.appendChild(a);
  a.click();
  a.remove();
}

function computeGlobalScore() {
  if (!CURRENT_USER) return 0;
  const s = CURRENT_USER.examScores;
  const examAvg = (s.windows + s.word + s.excel + s.ppt + s.access) / 5;
  return Math.round((examAvg * 0.7) + (CURRENT_USER.qcmScore * 0.3));
}

function updateTopStats() {
  if (!CURRENT_USER) return;
  document.getElementById('hdr-days').textContent = CURRENT_USER.studyDates.length;
  document.getElementById('hdr-global-pct').textContent = computeGlobalScore() + "%";
}

function openMainTab(tab, btnEl) {
  ACTIVE_TAB = tab;
  document.querySelectorAll('.nav-item').forEach(b => b.classList.remove('active'));
  if (btnEl) btnEl.classList.add('active');
  const ws = document.getElementById('main-workspace');

  if (tab === 'dashboard') renderDashboard(ws);
  else if (['windows', 'word', 'excel', 'ppt', 'access'].includes(tab)) renderSoftwareSpace(ws, tab);
  else if (tab === 'exam') renderFinalExamSpace(ws);
  else if (tab === 'qcm') renderQcmAndCertificate(ws);
  else if (tab === 'creator_mgr') renderCreatorManagerTab(ws);
}

function mgrAcceptCode() {
  const codeVal = document.getElementById('mgr-code-input').value.trim();
  const assignee = document.getElementById('mgr-assignee-input').value.trim() || "Élève autorisé";
  const fb = document.getElementById('mgr-feedback');
  if (!codeVal) {
    fb.style.color = "var(--danger)";
    fb.textContent = "⚠️ Saisissez d'abord un code (ex: 20262).";
    return;
  }
  const codes = JSON.parse(localStorage.getItem(STORAGE_CODES_KEY));
  if (codes[codeVal]) {
    fb.style.color = "var(--danger)";
    fb.textContent = `⚠️ Le code ${codeVal} existe déjà.`;
    return;
  }
  codes[codeVal] = { code: codeVal, assignedTo: assignee, claimedBy: null, claimedName: null, isFounder: false };
  localStorage.setItem(STORAGE_CODES_KEY, JSON.stringify(codes));
  renderCreatorManagerTab(document.getElementById('main-workspace'));

  document.getElementById('mgr-feedback').textContent = `✅ Code ${codeVal} accepté et activé pour ${assignee}.`;

function mgrGenerateNext() {
  const codes = JSON.parse(localStorage.getItem(STORAGE_CODES_KEY));
  let maxN = 1;
  Object.keys(codes).forEach(k => {
    const n = parseInt(k.replace("2026", ""), 10);
    if (!isNaN(n) && n > maxN) maxN = n;
  });
  const nextCode = "2026" + (maxN + 1);
  const assignee = document.getElementById('mgr-assignee-input').value.trim() || `Élève ${maxN}`;
  codes[nextCode] = { code: nextCode, assignedTo: assignee, claimedBy: null, claimedName: null, isFounder: false };
  localStorage.setItem(STORAGE_CODES_KEY, JSON.stringify(codes));
  renderCreatorManagerTab(document.getElementById('main-workspace'));
  document.getElementById('mgr-feedback').style.color = "var(--success)";
  document.getElementById('mgr-feedback').textContent = `✅ Code ${nextCode} généré et accepté pour ${assignee}.`;
}

function normalizeStr(s) {
  return s.toLowerCase().replace(/\s+/g, ' ').trim();
}

function validatePractice(sw, itemKey) {
  const input = document.getElementById('practice-input').value;
  const fb = document.getElementById('practice-feedback');
  if (!input.trim()) {
    fb.style.color = "var(--danger)";
    fb.textContent = "⚠️ Veuillez saisir votre réponse complète avant d'appuyer sur Entrée.";
    return;
  }

  const data = CURRICULUM[sw];
  const sub = ACTIVE_SW_SUB[sw];
  const idx = ACTIVE_SW_IDX[sw];
  const obj = sub === 'courses' ? data.courses[idx].exercise : data.cases[idx];

  const userClean = normalizeStr(input);
  const validList = [normalizeStr(obj.expected), ...(obj.alt || []).map(normalizeStr)];

  if (validList.includes(userClean)) {
    CURRENT_USER.completedItems[itemKey] = true;
    saveUserProgress();
    fb.style.color = "var(--success)";
    fb.textContent = "🎉 EXCELLENT ! Réponse exacte validée et sauvegardée automatiquement dans votre dossier.";
    setCoachMessage(`Bravo ${CURRENT_USER.prenom} ! Tu as validé "${sub === 'courses' ? data.courses[idx].title : data.cases[idx].title}".`);
    setTimeout(() => renderSoftwareSpace(document.getElementById('main-workspace'), sw), 900);
  } else {
    fb.style.color = "var(--danger)";
    fb.textContent = `❌ Réponse incorrecte ou incomplète. Vérifiez la syntaxe exacte (ex: signe =, guillemets, raccourci) et appuyez de nouveau sur Entrée.`;
  }
}

function gradeExamSoftware(sw) {
  const examData = CURRICULUM[sw].examTasks[0];
  let correct = 0;
  examData.questions.forEach((qObj, idx) => {
    const val = normalizeStr(document.getElementById(`exam-ans-${idx}`).value);
    if (val === normalizeStr(qObj.ans)) correct++;
  });
  const pct = Math.round((correct / examData.questions.length) * 100);
  CURRENT_USER.examScores[sw] = Math.max(CURRENT_USER.examScores[sw], pct);
  saveUserProgress();

  const resBox = document.getElementById('exam-result-box');
  resBox.innerHTML = `<div class="alert ${pct>=70?'alert-success':'alert-error'}" style="display:block;">
    Résultat de l'épreuve ${CURRICULUM[sw].name} : <strong>${correct} / ${examData.questions.length} bonnes réponses — Score : ${pct}%</strong> (Meilleur score sauvegardé : ${CURRENT_USER.examScores[sw]}%).
  </div>`;
  setCoachMessage(`Copie ${CURRICULUM[sw].name} corrigée : ${pct}% ! Ton score est sauvegardé.`);
}

function submitFinalQcm(e) {
  e.preventDefault();
  let good = 0;
  FINAL_QCM.forEach((item, idx) => {
    const checked = document.querySelector(`input[name="qcm_${idx}"]:checked`);
    if (checked && parseInt(checked.value, 10) === item.ans) good++;
  });
  CURRENT_USER.qcmScore = Math.round((good / FINAL_QCM.length) * 100);
  CURRENT_USER.qcmDone = true;
  saveUserProgress();
  renderQcmAndCertificate(document.getElementById('main-workspace'));
  setCoachMessage(`Félicitations ${CURRENT_USER.prenom} ! Tu as obtenu ${CURRENT_USER.qcmScore}% au QCM. Ton certificat avec Code QR (05 044 71 63) est prêt !`);
}

function updateCertSignature() {
  const val = document.getElementById('sig-custom-input').value.trim();
  if (!val) return;
  CURRENT_USER.founderSig = val;
  saveUserProgress();
  document.getElementById('cert-sig-display').textContent = val;
}

function setCoachMessage(txt) {
  document.getElementById('coach-msg').textContent = txt;
}

function speakCoach() {
  if (!('speechSynthesis' in window)) return;
  window.speechSynthesis.cancel();
  const u = new SpeechSynthesisUtterance(document.getElementById('coach-msg').textContent);
  u.lang = 'fr-FR';
  window.speechSynthesis.speak(u);
}
